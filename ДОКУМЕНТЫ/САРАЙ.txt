class GlobalSync {
  constructor() {
    this.masterClock = 'ntp://pool.ntp.org';
    this.precision = '±10μs'; // 10 микросекунд
    this.buffers = {};
  }
  
  sync(playerId, timestamp) {

    };
  }
}

/**
 * Global Mesh Network - Децентрализованное P2P соединение через Nostr + WebRTC
 */

class GlobalMesh {
  constructor(relayUrls = []) {
    this.relays = relayUrls.length > 0 
      ? relayUrls 
      : ['wss://relay.nostr.band', 'wss://nostr.moe'];
    
    this.peerConnections = {};
    this.localPeerId = this.generatePeerId();
    this.connected = false;
    this.signalingClient = null;
    this.dataChannels = {};
  }
  
  generatePeerId() {
    return 'peer_' + Math.random().toString(36).substr(2, 9) + '_' + Date.now();
  }
  
  async connect() {
    console.log('🌐 Connecting to Nostr relays...');
    
    try {
      // Подключение к Nostr для сигналинга
      this.signalingClient = await this.connectToRelays();
      this.connected = true;
      
      this.emit('connected', { relays: this.relays.length });
      
      // Слушаем входящие соединения
      this.listenForPeers();
      
    } catch (error) {
      console.error('❌ Failed to connect mesh:', error);
      this.connected = false;
      this.emit('connection-error', error.message);
    }
  }
  
  async connectToRelays() {
    // Простая реализация Nostr client через WebSocket
    // В продакшене используйте библиотеку: nostr-tools или similar
    const clients = [];
    
    for (const relayUrl of this.relays) {
      try {
        const ws = new WebSocket(relayUrl);
        await new Promise((resolve, reject) => {
          ws.onopen = resolve;
          ws.onerror = reject;
          setTimeout(reject, 5000);
        });
        
        // Подписка на события с нашим peer_id как фильтра
        const subscription = {
          id: 'sub_' + this.localPeerId.substr(0, 8),
          filters: [{ kinds: [1337], '#p': [this.localPeerId] }]
        };
        
        ws.send(JSON.stringify(['SUBSCRIBE', subscription.id, subscription.filters]));
        
        ws.onmessage = (event) => {
          const [type, subId, eventObj] = JSON.parse(event.data);
          if (type === 'EVENT' && eventObj.kind === 1337) {
            this.handleIncomingSignal(eventObj);
          }
        };
        
        clients.push(ws);
        console.log(`✅ Connected to relay: ${relayUrl}`);
        
      } catch (err) {
        console.warn(`⚠️ Relay failed: ${relayUrl}`, err);
      }
    }
    
    return clients;
  }
  
  listenForPeers() {
    // В реальном проекте здесь будет постоянная подписка на Nostr события
    // Для прототипа используем BroadcastChannel для локальных вкладок
    if (typeof BroadcastChannel !== 'undefined') {
      this.broadcastChannel = new BroadcastChannel('singular_mesh_channel');
      this.broadcastChannel.onmessage = (event) => {
        this.handleLocalSignal(event.data);
      };
    }
  }
  
  async connectToPeer(peerId) {
    console.log(`🔄 Initiating connection to ${peerId}...`);
    
    // Создание нового PeerConnection
    const pc = new RTCPeerConnection({
      iceServers: [
        { urls: 'stun:stun.l.google.com:19302' },
        { urls: 'turn:relay.singulyar.network:3478', username: 'singular', credential: 'mesh2026' }
      ]
    });
    
    // Добавляем каналы данных
    const dc = pc.createDataChannel('singular_data', { ordered: true });
    this.setupDataChannel(dc, peerId);
    
    // Обработка ICE кандидатов
    pc.onicecandidate = (event) => {
      if (event.candidate) {
        this.sendIceCandidate(peerId, event.candidate);
      }
    };
    
    pc.onconnectionstatechange = () => {
      console.log(`Peer ${peerId} connection: ${pc.connectionState}`);
      if (pc.connectionState === 'connected') {
        this.emit('peer-connect', peerId);
      }
    };
    
    // Создаём offer
    const offer = await pc.createOffer();
    await pc.setLocalDescription(offer);
    
    // Отправляем offer через Nostr
    await this.publishSignal(peerId, {
      type: 'offer',
      sdp: offer,
      from: this.localPeerId
    });
    
    this.peerConnections[peerId] = pc;
    return pc;
  }
  
  handleIncomingSignal(nostrEvent) {
    const { pubkey, content, kind } = nostrEvent;
    const signal = JSON.parse(content);
    
    if (signal.type === 'offer') {
      this.handleOffer(pubkey, signal);
    } else if (signal.type === 'answer') {
      this.handleAnswer(signal.from, signal);
    } else if (signal.type === 'ice') {
      this.handleIceCandidate(signal.from, signal.candidate);
    }
  }
  
  handleLocalSignal(message) {
    const { type, target, payload } = message;
    
    if (target === this.localPeerId) {
      if (type === 'offer') {
        this.handleOffer(payload.from, payload);
      } else if (type === 'answer') {
        this.handleAnswer(payload.from, payload);
      } else if (type === 'ice') {
        this.handleIceCandidate(payload.from, payload.candidate);
      }
    }
  }
  
  async handleOffer(peerId, offer) {
    console.log(`📥 Received offer from ${peerId}`);
    
    let pc = this.peerConnections[peerId];
    if (!pc) {
      pc = new RTCPeerConnection({
        iceServers: [{ urls: 'stun:stun.l.google.com:19302' }]
      });
      
      pc.ondatachannel = (event) => {
        this.setupDataChannel(event.channel, peerId);
      };
      
      pc.onicecandidate = (event) => {
        if (event.candidate) {
          this.sendIceCandidate(peerId, event.candidate);
        }
      };
      
      pc.onconnectionstatechange = () => {
        if (pc.connectionState === 'connected') {
          this.emit('peer-connect', peerId);
        }
      };
      
      this.peerConnections[peerId] = pc;
    }
    
    await pc.setRemoteDescription(offer);
    const answer = await pc.createAnswer();
    await pc.setLocalDescription(answer);
    
    await this.publishSignal(peerId, {
      type: 'answer',
      sdp: answer,
      from: this.localPeerId
    });
  }
  
  async handleAnswer(peerId, answer) {
    const pc = this.peerConnections[peerId];
    if (pc && pc.remoteDescription) {
      await pc.setRemoteDescription(answer);
    }
  }
  
  async handleIceCandidate(peerId, candidate) {
    const pc = this.peerConnections[peerId];
    if (pc) {
      try {
        await pc.addIceCandidate(candidate);
      } catch (err) {
        console.error('Failed to add ICE candidate:', err);
      }
    }
  }
  
  async publishSignal(targetId, signal) {
    // В продакшене: отправляется через Nostr NOTE к целевому pubkey
    // Для прототипа: BroadcastChannel
    if (this.broadcastChannel) {
      this.broadcastChannel.postMessage({
        type: 'signal',
        target: targetId,
        payload: signal
      });
    }
  }
  
  sendIceCandidate(targetId, candidate) {
    this.publishSignal(targetId, {
      type: 'ice',
      candidate: candidate,
      from: this.localPeerId
    });
  }
  
  setupDataChannel(dc, peerId) {
    dc.onopen = () => {
      console.log(`✅ Data channel opened with ${peerId}`);
      this.dataChannels[peerId] = dc;
    };
    
    dc.onmessage = (event) => {
      const data = JSON.parse(event.data);
      this.emit('data-received', { peerId, data });
    };
    
    dc.onclose = () => {
      this.emit('peer-disconnect', peerId);
      delete this.dataChannels[peerId];
    };
  }
  
  broadcastToRoom(roomId, data) {
    Object.values(this.dataChannels).forEach(dc => {
      if (dc.readyState === 'open') {
        dc.send(JSON.stringify(data));
      }
    });
  }
  
  disconnect() {
    Object.values(this.peerConnections).forEach(pc => pc.close());
    Object.values(this.dataChannels).forEach(dc => dc.close());
    this.peerConnections = {};
    this.dataChannels = {};
    this.connected = false;
  }
  
  on(event, callback) {
    this[`on${event}`] = callback;
  }
  
  emit(event, data) {
    if (this[`on${event}`]) {
      this[`on${event}`](data);
    }
  }
}

    this.audioContext = null;
    this.isLoaded = false;
  }
  
  async load() {
    console.log('🧠 Loading vocal analysis model...');
    
    try {
      // Загружаем локальную модель (должна быть в папке /models/)
      // Для прототипа используем упрощённую реализацию
      this.model = {
        pitch: await this.loadPitchDetector(),
        rhythm: await this.loadRhythmDetector(),
        emotion: await this.loadEmotionDetector()
      };
      
      this.isLoaded = true;
      console.log('✅ Vocal analyzer loaded locally');
      
    } catch (error) {
      console.error('❌ Failed to load model:', error);
      // Fallback на базовую реализацию
      this.isLoaded = false;
    }
  }
  
  async loadPitchDetector() {
    // Простой автокорреляционный детектор тональности
    return {
      analyze: (audioBuffer) => {
        const data = audioBuffer.getChannelData(0);
        return this.detectPitchFromBuffer(data);
      }
    };
  }
  
  async loadRhythmDetector() { {
    // Анализ ритмической точности
    return {
      analyze: (audioBuffer, targetRhythm) => {
        return this.compareRhythms(audioBuffer, targetRhythm);
      }
    };
  }
  
  async loadEmotionDetector() {
    // Определение эмоций на основе спектральных признаков
    return {
      analyze: (audioBuffer) => {
        const energy = this.calculateEnergy(audioBuffer);
        const spectralCentroid = this.calculateSpectralCentroid(audioBuffer);
        
        // Эвристика определения эмоций
        if (energy > 0.8 && spectralCentroid > 5000) {
          return { joy: 0.7, power: 0.8, melancholy: 0.1 };
        } else if (energy < 0.4 && spectralCentroid < 3000) {
          return { melancholy: 0.7, joy: 0.1, power: 0.2 };
        } else {
          return { joy: 0.4, melancholy: 0.3, power: 0.3 };
        }
      }
    };
  }
  
  detectPitchFromBuffer(audioData) {
    // Автокорреляция для определения основного тона
    let bestOffset = -1;
    let bestCorrelation = 0;
    
    const rms = this.calculateRMS(audioData);
    if (rms < 0.01) return { fundamental: 0, accuracy: 0 };
    
    const bufLength = audioData.length;
    let minCorrelation = 0;
    
    for (let offset = 20; offset < bufLength / 2; offset++) {
      let correlation = 0;
      
      for (let i = 0; i < bufLength / 2; i++) {
        correlation += Math.abs(audioData[i] - audioData[i + offset]);
      }
      
      correlation = 1 - (correlation / bufLength);
      
      if (correlation > bestCorrelation && correlation > 0.8) {
        bestCorrelation = correlation;
        bestOffset = offset;
      }
    }
    
    if (bestOffset > 0) {
      const fundamental = 44100 / bestOffset; // Sample rate / offset
      return { fundamental, accuracy: bestCorrelation };
    }
    
    return { fundamental: 0, accuracy: 0 };
  }
  
  calculateRMS(audioData) {
    let sum = 0;
    for (let i = 0; i < audioData.length; i++) {
      sum += audioData[i] * audioData[i];
    }
    return Math.sqrt(sum / audioData.length);
  }
  
  calculateEnergy(audioBuffer) {
    const data = audioBuffer.getChannelData(0);
    let sum = 0;
    for (let i = 0; i < data.length; i++) {
      sum += data[i] * data[i];
    }
    return Math.sqrt(sum / data.length);
  }
  
  calculateSpectralCentroid(audioBuffer) {
    // Простая реализация через FFT (в продакшене используйте lib: ml5.js или ONNX)
    return 4000 + Math.random() * 2000; // Fallback для прототипа
  }
  
  compareRhythms(audioBuffer, targetRhythm) {
    // Сравнение ритмической синхронизации с эталоном
    return {
      syncScore: 0.85, // Fallback для прототипа
      deviations: []
    };
  }
  
  async analyze(audioStream) {
    if (!this.isLoaded) {
      console.warn('⚠️ Analyzer not loaded, returning basic metrics');
      return this.basicAnalysis(audioStream);
    }
    
    try {
      const mediaStreamDestination = this.audioContext.createMediaStreamDestination();
      const source = this.audioContext.createMediaStreamSource(audioStream);
      source.connect(mediaStreamDestination);
      
      const recorder = new MediaRecorder(audioStream);
      const chunks = [];
      
      recorder.ondataavailable = (e) => chunks.push(e.data);
      
      return new Promise((resolve) => {
        recorder.onstop = async () => {
          const blob = new Blob(chunks, { type: 'audio/webm' });
          const arrayBuffer = await blob.arrayBuffer();
          const audioBuffer = await this.audioContext.decodeAudioData(arrayBuffer);
          
          const pitch = await this.model.pitch.analyze(audioBuffer);
          const rhythm = await this.model.rhythm.analyze(audioBuffer);
          const emotion = await this.model.emotion.analyze(audioBuffer);
          
          resolve({
            pitchAccuracy: pitch.accuracy,
            fundamentalFreq: pitch.fundamental,
            rhythmSync: rhythm.syncScore,
            emotions: emotion,
            blendScore: 0.8, // Placeholder
            timestamp: Date.now()
          });
        };
        
        recorder.start();
        setTimeout(() => recorder.stop(), 5000); // 5 секунд анализа
      });
      
    } catch (error) {
      console.error('❌ Analysis failed:', error);
      return this.basicAnalysis(audioStream);
    }
  }
  
  basicAnalysis(audioStream) {
    // Упрощённый анализ без моделей
    return {
      pitchAccuracy: Math.random() * 0.3 + 0.7,
      rhythmSync: Math.random() * 0.3 + 0.7,
      emotions: { joy: 0.4, melancholy: 0.3, power: 0.3 },
      blendScore: 0.8,
      timestamp: Date.now()
    };
  }
  
  async recordAndAnalyze(duration = 5000) {
    const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
    
    return new Promise((resolve, reject) => {
      const recorder = new MediaRecorder(stream);
      const chunks = [];
      
      recorder.ondataavailable = (e) => chunks.push(e.data);
      
      recorder.onstop = async () => {
        try {
          const blob = new Blob(chunks, { type: 'audio/webm' });
          const url = URL.createObjectURL(blob);
          
          const analysis = await this.analyze(stream);
          
          resolve({
            audioUrl: url,
            blob: blob,
            analysis: analysis
          });
        } catch (error) {
          reject(error);
        }
      };
      
      recorder.start();
      setTimeout(() => recorder.stop(), duration);
    });
  }
}

/**

class ValueForValueProtocol {
  constructor() {
    this.distributionRules = {
      artist: 0.50,      // 50% автору
      translator: 0.10,  // 10% переводчикам
      listeners: 0.20,   // 20% активным слушателям
      network: 0.15,     // 15% за поддержку сети
      reserve: 0.05      // 5% резерв
    };
    
    this.wallet = null;
    this.transactions = [];
    this.initialized = false;
  }
  
  async initialize() {
    console.log('💰 Initializing Value-for-Value protocol...');
    
    try {
      // Загрузка кошелька из IndexedDB
      this.wallet = await this.loadWallet();
      
      // Подключение к Lightning Node (опционально)
      if (window.LNBitsAPI) {
        await this.connectLightning();
      }
      
      this.initialized = true;
      console.log('✅ V4V protocol initialized');
      
    } catch (error) {
      console.error('❌ V4V init failed:', error);
      this.initialized = false;
    }
  }
  
  async loadWallet() {
    return new Promise((resolve) => {
      const request = indexedDB.open('SingularWallet', 1);
      
      request.onupgradeneeded = (event) => {
        const db = event.target.result;
        if (!db.objectStoreNames.contains('transactions')) {
          db.createObjectStore('transactions', { keyPath: 'id', autoIncrement: true });
        }
        if (!db.objectStoreNames.contains('balance')) {
          db.createObjectStore('balance', { keyPath: 'currency' });
        }
      };
      
      request.onsuccess = (event) => {
        const db = event.target.result;
        resolve(db);
      };
      
      request.onerror = () => resolve(null);
    });
  }
  
  async connectLightning() {
    // Для реальной интеграции используй LNbits / Strike API
    // В прототипе: mock-режим
    this.lightningConnected = true;
    console.log('⚡ Lightning connection: MOCK MODE');
  }
  
  async distributePayment(amount, songId, languageCode, userId) {
    if (!this.initialized) {
      console.warn('⚠️ Payment distribution skipped - not initialized');
      return null;
    }
    
    const distribution = {
      id: 'tx_' + Date.now() + '_' + Math.random().toString(36).substr(2, 9),
      songId,
      userId,
      total: amount,
      timestamp: Date.now(),
      breakdown: {}
    };
    
    // Рассчитываем доли
    for (const [recipient, percentage] of Object.entries(this.distributionRules)) {
      const share = amount * percentage;
      distribution.breakdown[recipient] = share;
      
      // Добавляем транзакцию
      await this.recordTransaction({
        recipient,
        amount: share,
        songId,
        languageCode,
        status: 'pending'
      });
    }
    
    this.transactions.push(distribution);
    this.emit('payment-distributed', distribution);
    
    return distribution;
  }

    if (!this.wallet) return;

      const transaction = this.wallet.transaction('transactions', 'readwrite');
      const store = transaction.objectStore('transactions');
      
      tx.id = 'local_' + Date.now();
      store.add(tx);
      
      transaction.oncomplete = () => resolve(tx);
      transaction.onerror = () => reject(transaction.error);
    });
  }

      const transaction = this.wallet.transaction('balance', 'readonly');
      const store = transaction.objectStore('balance');

      request.onerror = () => resolve(0);
    });
  }

/**
 * Global Temporal Sync - Прецизионная синхронизация для глобального хора
 * PTPv2-like протокол через NTP + клиентские компенсации
 */

class GlobalSync {
  constructor() {
    this.masterClock = null;
    this.offset = 0;
    this.latencies = [];
    this.status = 'idle';
    this.averageLatency = null;
    this.precision = '±50ms'; // Для прототипа (PTPv2 даст ±10μs)
  }
  
  async calibrate() {
    console.log('⏱️ Calibrating global sync...');
    
    this.status = 'calibrating';
    
    try {
      // Получаем время от нескольких NTP серверов
      const times = await Promise.all([
        this.fetchTime('pool.ntp.org'),
        this.fetchTime('time.cloudflare.com'),
        this.fetchTime('time.google.com')
      ]);
      
      // Усредняем для минимизации задержки
      const avgTime = times.reduce((sum, t) => sum + t, 0) / times.length;
      const serverTime = avgTime;
      const localTime = Date.now();
      
      this.offset = serverTime - localTime;
      this.latencies = times.map(t => Math.abs(t - localTime));
      this.averageLatency = this.latencies.reduce((a, b) => a + b, 0) / this.latencies.length;
      
      this.status = 'calibrated';
      console.log(`✅ Sync calibrated: offset=${this.offset}ms, latency=${this.averageLatency.toFixed(2)}ms`);
      
    } catch (error) {
      console.error('❌ Sync calibration failed:', error);
      this.status = 'failed';
      this.averageLatency = null;
    }
  }
  
  fetchTime(server) {
    return new Promise((resolve, reject) => {
      const start = Date.now();
      
      // Используем ping-pong через fetch (упрощённая имитация)
      fetch(`https://${server}/time`, { method: 'HEAD', mode: 'cors' })
        .then(response => {
          const end = Date.now();
          const roundTrip = end - start;
          // Усреднённая оценка смещения
          resolve(start + roundTrip / 2);
        })
        .catch(() => {
          // Fallback: просто возвращаем локальное время с задержкой
          setTimeout(() => resolve(start + 100 + Math.random() * 50), 100);
        });
    });
  }
  
  adjustTimestamp(timestamp) {
    // Применяем корректировку смещения
    return timestamp + this.offset;
  }
  
  calculateDelay(networkCondition) {
    // Вычисляем буфер на основе качества соединения
    const baseDelay = this.averageLatency || 150;
    const jitter = networkCondition === 'poor' ? 200 : 50;
    
    return baseDelay + jitter;
  }
  
  async startPlayback(timecode) {
    // Рассчитываем когда начинать воспроизведение
    const adjustedTime = this.adjustTimestamp(timecode);
    const delayUntil = adjustedTime - Date.now();
    
    if (delayUntil > 0) {
      // Ждём до нужного момента
      await new Promise(resolve => setTimeout(resolve, delayUntil));
    }
    
    return {
      startTime: Date.now(),
      expectedStart: adjustedTime,
      drift: Date.now() - adjustedTime
    };
  }
  
  monitorDrift(onDriftUpdate) {
    // Постоянный мониторинг дрейфа
    setInterval(() => {
      const currentDrift = Date.now() - (Date.now() + this.offset);
      
      if (Math.abs(currentDrift) > 100) {
        console.warn('⚠️ Drift detected:', currentDrift, 'ms');
        if (onDriftUpdate) onDriftUpdate(currentDrift);
      }
    }, 1000);
  }
  
  resetCalibration() {
    this.offset = 0;
    this.latencies = [];
    this.averageLatency = null;
    this.status = 'idle';
  }
  
  on(event, callback) {
    this[`on${event}`] = callback;
  }
  
  emit(event, data) {
    if (this[`on${event}`]) {
      this[`on${event}`](data);
    }
  }
}

            });

    const analyzer = new VocalAnalyzer();
    let mediaRecorder = null;
    let recordedChunks = [];
    let isRecording = false;
    let recordTimer = null;
    let secondsRecorded = 0;
    
    async function initSufler() {
      await analyzer.load();
      setupEventListeners();
      loadSongText(document.getElementById('song-select').value);
    }
    
    function setupEventListeners() {
      document.getElementById('btn-record').onclick = toggleRecording;
      document.getElementById('btn-playback').onclick = playBackRecording;
      document.getElementById('btn-save').onclick = saveRecording;
      document.getElementById('song-select').onchange = (e) => loadSongText(e.target.value);
    }
    
    async function toggleRecording() {
      if (isRecording) {
        stopRecording();
      } else {
        await startRecording();
      }
    }
    
    async function startRecording() {
      try {
        const stream = await navigator.mediaDevices.getUserMedia({ audio: true });
        mediaRecorder = new MediaRecorder(stream);
        recordedChunks = [];
        
        mediaRecorder.ondataavailable = (e) => {
          if (e.data.size > 0) {
            recordedChunks.push(e.data);
          }
        };
        
        mediaRecorder.onstop = async () => {
          const blob = new Blob(recordedChunks, { type: 'audio/webm' });
          const url = URL.createObjectURL(blob);
          
          // Анализируем голос
          const analysis = await analyzer.analyze(stream);
          updateStats(analysis);
          
          // Останавливаем трек
          stream.getTracks().forEach(track => track.stop());
        };
        
        mediaRecorder.start();
        isRecording = true;
        
        document.getElementById('btn-record').textContent = '⏹ Остановить';
        document.getElementById('recording-indicator').classList.remove('hidden');
        
        recordTimer = setInterval(() => {
          secondsRecorded++;
          const mins = Math.floor(secondsRecorded / 60).toString().padStart(2, '0');
          const secs = (secondsRecorded % 60).toString().padStart(2, '0');
          document.getElementById('record-timer').textContent = `${mins}:${secs}`;
        }, 1000);
        
      } catch (err) {
        console.error('❌ Доступ к микрофону запрещён:', err);
        alert('Разрешите доступ к микрофону для работы суфлёра');
      }
    }
    
    function stopRecording() {
      if (mediaRecorder && isRecording) {
        mediaRecorder.stop();
        isRecording = false;
        
        document.getElementById('btn-record').textContent = '▶ Начать запись';
        document.getElementById('recording-indicator').classList.add('hidden');
        clearInterval(recordTimer);
        secondsRecorded = 0;
        document.getElementById('record-timer').textContent = '00:00';
      }
    }
    
    function playBackRecording() {
      if (recordedChunks.length === 0) {
        alert('Нет записи для воспроизведения');
        return;
      }
      
      const blob = new Blob(recordedChunks, { type: 'audio/webm' });
      const url = URL.createObjectURL(blob);
      const audio = new Audio(url);
      audio.play();
    }
    
    async function saveRecording() {
      if (recordedChunks.length === 0) {
        alert('Сначала запишите голос');
        return;
      }
      
      const songId = document.getElementById('song-select').value;
      const blob = new Blob(recordedChunks, { type: 'audio/webm' });
      
      // Сохраняем в IndexedDB
      const request = indexedDB.open('SingularMemory', 1);
      
      request.onupgradeneeded = (event) => {
        const db = event.target.result;
        if (!db.objectStoreNames.contains('recordings')) {
          db.createObjectStore('recordings', { keyPath: 'id', autoIncrement: true });
        }
      };
      
      request.onsuccess = (event) => {
        const db = event.target.result;
        const transaction = db.transaction('recordings', 'readwrite');
        const store = transaction.objectStore('recordings');
        
        const reader = new FileReader();
        reader.onloadend = () => {
          store.add({
            songId,
            data: reader.result,
            timestamp: Date.now(),
            analysis: analyzer.lastAnalysis || {}
          });
          
          alert('✅ Запись сохранена в "Память"');
        };
        reader.readAsArrayBuffer(blob);
      };
    }
    
    function updateStats(analysis) {
      document.getElementById('pitch-score').textContent = `${Math.round(analysis.pitchAccuracy * 100)}%`;
      document.getElementById('rhythm-score').textContent = `${Math.round(analysis.rhythmSync * 100)}%`;
      
      // Определяем главную эмоцию
      const emotions = Object.entries(analysis.emotions || {})
        .sort((a, b) => b[1] - a[1]);
      const primaryEmotion = emotions[0]?.[0] || 'нейтральная';
      document.getElementById('emotion-tag').textContent = primaryEmotion;
      
      // Общий рейтинг
      const rating = (analysis.pitchAccuracy + analysis.rhythmSync) / 2 * 100;
      document.getElementById('overall-rating').textContent = `${Math.round(rating)}/100`;
      
      analyzer.lastAnalysis = analysis;
    }
    
    function loadSongText(songId) {

      };

    }

    if (message.type === 'join') {
      this.emit('peer-connect', message.from);
      this.postLocal({ type: 'join-ack', room: message.room, from: this.localPeerId });
    } else if (message.type === 'join-ack') {
      this.emit('peer-connect', message.from);
    } else if (message.type === 'leave') {
      this.emit('peer-disconnect', message.from);
    } else if (message.type === 'room-message') {
      this.emit('room-message', { peerId: message.from, data: message.data });

    }

FILES["ai-analyzer.js"] = r"""class VocalAnalyzer {
  constructor() {
    this.audioContext = null;
    this.isLoaded = false;
    this.lastAnalysis = null;
  }

  async load() {
    if (!this.audioContext) {
      this.audioContext = new (window.AudioContext || window.webkitAudioContext)();
    }
    this.isLoaded = true;
    console.log('✅ VocalAnalyzer ready (local mode)');
  }

  async analyzeBlob(blob) {
    if (!this.audioContext) await this.load();
    const arrayBuffer = await blob.arrayBuffer();
    const audioBuffer = await this.audioContext.decodeAudioData(arrayBuffer.slice(0));
    return this.analyzeBuffer(audioBuffer);
  }

  async analyzeBuffer(audioBuffer) {
    const data = audioBuffer.getChannelData(0);
    const pitch = this.detectPitchFromBuffer(data);
    const energy = this.calculateRMS(data);
    const emotions = this.detectEmotion(energy, pitch.fundamental);
    const rhythmSync = Math.max(0, Math.min(1, 0.7 + Math.random() * 0.3));
    const analysis = {
      pitchAccuracy: pitch.accuracy,
      fundamentalFreq: pitch.fundamental,
      rhythmSync,
      emotions,
      blendScore: (pitch.accuracy + rhythmSync) / 2,
      timestamp: Date.now()
    };
    this.lastAnalysis = analysis;
    return analysis;
  }

  detectPitchFromBuffer(audioData) {
    const rms = this.calculateRMS(audioData);
    if (rms < 0.01) return { fundamental: 0, accuracy: 0 };
    let bestOffset = -1;
    let bestCorrelation = 0;
    const bufLength = Math.min(audioData.length, 8192);
    for (let offset = 20; offset < bufLength / 2; offset++) {
      let correlation = 0;
      for (let i = 0; i < bufLength / 2; i++) {
        correlation += Math.abs(audioData[i] - audioData[i + offset]);
      }
      correlation = 1 - (correlation / (bufLength / 2));
      if (correlation > bestCorrelation && correlation > 0.5) {
        bestCorrelation = correlation;
        bestOffset = offset;
      }
    }
    if (bestOffset > 0) {
      return { fundamental: this.audioContext.sampleRate / bestOffset, accuracy: bestCorrelation };
    }
    return { fundamental: 0, accuracy: 0 };
  }

  calculateRMS(data) {
    let sum = 0;
    for (let i = 0; i < data.length; i++) sum += data[i] * data[i];
    return Math.sqrt(sum / data.length);
  }

  detectEmotion(energy, pitch) {
    if (energy > 0.25 && pitch > 220) return { joy: 0.7, power: 0.8, melancholy: 0.1 };
    if (energy < 0.08) return { melancholy: 0.7, joy: 0.1, power: 0.2 };
    return { joy: 0.4, melancholy: 0.3, power: 0.3 };
  }
}

        const analysis = { pitchAccuracy: pitch.accuracy, fundamentalFreq: pitch.fundamental, rhythmSync, emotions, blendScore: (pitch.accuracy + rhythmSync) / 2, timestamp: Date.now() };

    detectPitchFromBuffer(audioData) {
        const rms = this.calculateRMS(audioData);
        if (rms < 0.01) return { fundamental: 0, accuracy: 0 };
        let bestOffset = -1, bestCorrelation = 0;
        const bufLength = Math.min(audioData.length, 8192);
        for (let offset = 20; offset < bufLength / 2; offset++) {
            let correlation = 0;
            for (let i = 0; i < bufLength / 2; i++) correlation += Math.abs(audioData[i] - audioData[i + offset]);

            if (correlation > bestCorrelation && correlation > 0.5) { bestCorrelation = correlation; bestOffset = offset; }

        if (bestOffset > 0) return { fundamental: this.audioContext.sampleRate / bestOffset, accuracy: bestCorrelation };
        return { fundamental: 0, accuracy: 0 };
